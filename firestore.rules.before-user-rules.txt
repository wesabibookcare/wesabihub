rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // -------------------------------------------------------------------------
    // GLOBAL HELPERS
    // -------------------------------------------------------------------------

    function isSignedIn() {
      return request.auth != null;
    }

    function isEmailVerified() {
      return isSignedIn() && request.auth.token.email_verified == true;
    }

    function isOwner(userId) {
      return isSignedIn() && request.auth.uid == userId;
    }

    function incoming() {
      return request.resource.data;
    }

    function existing() {
      return resource.data;
    }

    function getUserData(uid) {
      return get(/databases/$(database)/documents/users/$(uid)).data;
    }

    function hasRole(role) {
      return isSignedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && (
          ('roles' in getUserData(request.auth.uid) && getUserData(request.auth.uid).roles.hasAny([role])) ||
          ('role' in getUserData(request.auth.uid) && getUserData(request.auth.uid).role == role)
        );
    }

    function hasAnyRole(roles) {
      return isSignedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && (
          ('roles' in getUserData(request.auth.uid) && getUserData(request.auth.uid).roles.hasAny(roles)) ||
          ('role' in getUserData(request.auth.uid) && getUserData(request.auth.uid).role in roles)
        );
    }

    function isAdmin() {
      return hasRole('SUPER_ADMIN') || hasRole('SECURITY_ADMIN') || hasRole('OPERATIONS_ADMIN');
    }

    function isPlatformAdmin() {
      return hasAnyRole(['SUPER_ADMIN', 'OPERATIONS_MANAGER', 'OPERATIONS_ADMIN', 'SECURITY_ADMIN', 'SUPPORT_ADMIN', 'FINANCE_ADMIN', 'VERIFICATION_ADMIN', 'DISPUTE_ADMIN']);
    }

    function isStaff() {
      return hasAnyRole(['SUPER_ADMIN', 'OPERATIONS_MANAGER', 'OPERATIONS_ADMIN', 'SECURITY_ADMIN', 'SUPPORT_ADMIN', 'FINANCE_ADMIN', 'VERIFICATION_ADMIN', 'DISPUTE_ADMIN', 'SUPPORT_OFFICER', 'VERIFICATION_OFFICER', 'FINANCE_OFFICER']);
    }

    function isValidId(id) {
      return id is string && id.size() <= 128 && id.matches('^[a-zA-Z0-9_\\-]+$');
    }

    function isImmutable(field) {
      return !(field in incoming().diff(existing()).affectedKeys());
    }

    // -------------------------------------------------------------------------
    // VALIDATION HELPERS
    // -------------------------------------------------------------------------

    function isBootstrapActive() {
      return !exists(/databases/$(database)/documents/systemSettings/global) ||
             !('isSuperAdminBootstrapped' in get(/databases/$(database)/documents/systemSettings/global).data) ||
             get(/databases/$(database)/documents/systemSettings/global).data.isSuperAdminBootstrapped != true;
    }

    function isValidUser(data, isAdminUpdate) {
      return (isAdminUpdate || data.uid == request.auth.uid)
        && data.email is string && data.email.size() <= 256
        && (!('roles' in data) || data.roles is list)
        && (!('status' in data) || (data.status is string && data.status.size() <= 64));
    }

    function isVerifiedMerchant(uid) {
      return exists(/databases/$(database)/documents/users/$(uid)) && (
        get(/databases/$(database)/documents/users/$(uid)).data.role == 'MERCHANT' ||
        ('roles' in get(/databases/$(database)/documents/users/$(uid)).data && 'MERCHANT' in get(/databases/$(database)/documents/users/$(uid)).data.roles)
      ) && (
        get(/databases/$(database)/documents/users/$(uid)).data.status in ['APPROVED', 'ACTIVE'] ||
        ('verificationStatus' in get(/databases/$(database)/documents/users/$(uid)).data &&
         get(/databases/$(database)/documents/users/$(uid)).data.verificationStatus.kyc == true)
      );
    }

    function isValidShipment(data) {
      return data.trackingNumber is string && data.trackingNumber.size() <= 32
        && data.status in ['DRAFT', 'AWAITING_PAYMENT', 'PAYMENT_CONFIRMED', 'AWAITING_DROP_OFF', 'RECEIVED_AT_ORIGIN', 'AWAITING_DISPATCH', 'IN_TRANSIT', 'TRANSFERRED_BETWEEN_POINTS', 'ARRIVED_AT_DESTINATION', 'READY_FOR_PICKUP', 'COLLECTED', 'DELIVERED', 'COMPLETED', 'CANCELLED', 'EXPIRED', 'LOST', 'DAMAGED', 'RETURNED', 'REFUND_PENDING', 'DISPUTED', 'ARCHIVED']
        && data.pricing.total >= 0;
    }

    // -------------------------------------------------------------------------
    // COLLECTIONS
    // -------------------------------------------------------------------------

    match /merchant_customers {
      allow read: if isSignedIn() && (hasRole('MERCHANT') || isStaff());
      match /{customerId} {
        allow get: if isSignedIn() && (resource.data.merchantId == request.auth.uid || isStaff());
        allow create: if isSignedIn() && (incoming().merchantId == request.auth.uid || isStaff());
        allow update: if isSignedIn() && (
          (existing().merchantId == request.auth.uid || isStaff()) &&
          (incoming().merchantId == existing().merchantId)
        );
        allow delete: if isSignedIn() && (existing().merchantId == request.auth.uid || isStaff());
      }
    }

    match /users/{userId} {
      allow get: if isSignedIn() && isValidId(userId);
      allow list: if isSignedIn();
      allow create: if (isOwner(userId) && isValidId(userId) && isValidUser(incoming(), false) && (
                        isAdmin() ||
                        (isBootstrapActive() && (('roles' in incoming() && 'SUPER_ADMIN' in incoming().roles) || ('role' in incoming() && incoming().role == 'SUPER_ADMIN'))) ||
                        ((!('roles' in incoming()) || !incoming().roles.hasAny(['SUPER_ADMIN', 'SUPPORT_OFFICER', 'VERIFICATION_OFFICER', 'OPERATIONS_MANAGER', 'FINANCE_OFFICER', 'DISPUTE_ADMIN', 'SUPPORT_ADMIN', 'OPERATIONS_ADMIN', 'VERIFICATION_ADMIN', 'SECURITY_ADMIN', 'FINANCE_ADMIN'])) &&
                         (!('role' in incoming()) || !(incoming().role in ['SUPER_ADMIN', 'SUPPORT_OFFICER', 'VERIFICATION_OFFICER', 'OPERATIONS_MANAGER', 'FINANCE_OFFICER', 'DISPUTE_ADMIN', 'SUPPORT_ADMIN', 'OPERATIONS_ADMIN', 'VERIFICATION_ADMIN', 'SECURITY_ADMIN', 'FINANCE_ADMIN'])))
                      )) || (isAdmin() && isValidId(userId));

      // PHASE A2: Owner updates are RESTRICTED to safe, non-privileged fields only.
      // Users cannot modify role/roles/status or any field that affects authorization.
      allow update: if (isOwner(userId) && isValidId(userId) &&
                        incoming().diff(existing()).affectedKeys().hasOnly([
                          'displayName', 'photoURL', 'photoUrl', 'phoneNumber', 'phone',
                          'country', 'businessType', 'telegramChatId', 'updatedAt'
                        ]))
                    || (isAdmin() && isValidId(userId) && isValidUser(incoming(), true));

      allow delete: if isAdmin() && isValidId(userId) && !('SUPER_ADMIN' in existing().roles); // Super Admin cannot be deleted
    }

    match /profile_changes/{id} {
      allow read, write: if isSignedIn() && (resource.data.userId == request.auth.uid || isStaff());
    }

    match /shipments/{shipmentId} {
      allow list: if isSignedIn() && (
        isStaff() ||
        hasAnyRole(['CENTER_STAFF', 'CENTER_OWNER', 'DRIVER']) ||
        (request.auth.uid != null)
      );
      allow get: if isSignedIn() && isValidId(shipmentId) && (
        isOwner(resource.data.senderId) ||
        resource.data.recipientInfo.email == request.auth.token.email ||
        (resource.data.recipientInfo.phone != null && exists(/databases/$(database)/documents/users/$(request.auth.uid)) && (get(/databases/$(database)/documents/users/$(request.auth.uid)).data.phoneNumber == resource.data.recipientInfo.phone || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.phone == resource.data.recipientInfo.phone)) ||
        isStaff() ||
        hasAnyRole(['CENTER_STAFF', 'CENTER_OWNER', 'DRIVER'])
      );
      allow create: if isSignedIn() && isValidId(shipmentId) && isValidShipment(incoming()) && (
        // Case 1: Verified Merchant creating their own shipment
        (incoming().senderId == request.auth.uid && isVerifiedMerchant(request.auth.uid)) ||
        // Case 2: Hub Staff/Owner creating on behalf of a verified Merchant
        (hasAnyRole(['CENTER_STAFF', 'CENTER_OWNER']) && isVerifiedMerchant(incoming().senderId)) ||
        // Case 3: Admin / Staff
        isStaff()
      );
      allow update: if isSignedIn() && isValidId(shipmentId) && isValidShipment(incoming()) && (
        (isOwner(existing().senderId) && existing().status in ['DRAFT', 'AWAITING_PAYMENT'] && incoming().diff(existing()).affectedKeys().hasOnly(['recipientInfo', 'dimensions', 'status', 'updatedAt']))
        || ((isStaff() || hasAnyRole(['CENTER_STAFF', 'CENTER_OWNER', 'DRIVER'])) && (
          // Critical Security Block: Clients cannot directly authorize PINs or transition to COLLECTED/DELIVERED states
          (!('pickupPinVerified' in incoming()) || incoming().pickupPinVerified == existing().pickupPinVerified || incoming().pickupPinVerified == false) &&
          !(incoming().status in ['COLLECTED', 'DELIVERED'])
        ))
      );
    }

    match /addresses/{addressId} {
      allow list: if isSignedIn() && resource.data.userId == request.auth.uid;
      allow get: if isSignedIn() && resource.data.userId == request.auth.uid;
      allow create: if isSignedIn() && incoming().userId == request.auth.uid;
      allow update: if isSignedIn() && existing().userId == request.auth.uid;
      allow delete: if isSignedIn() && existing().userId == request.auth.uid;
    }

    match /trackingEvents/{eventId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn() && (isStaff() || hasAnyRole(['CENTER_STAFF', 'CENTER_OWNER', 'DRIVER', 'CUSTOMER', 'MERCHANT']));
    }

    match /parcelTransfers/{transferId} {
      allow read: if isSignedIn() && hasAnyRole(['CENTER_STAFF', 'CENTER_OWNER', 'DRIVER', 'SUPER_ADMIN', 'OPERATIONS_MANAGER']);
      allow create, update: if isSignedIn() && hasAnyRole(['CENTER_STAFF', 'CENTER_OWNER', 'DRIVER', 'SUPER_ADMIN', 'OPERATIONS_MANAGER']);
    }

    match /parcelHistory/{historyId} {
      allow read: if isSignedIn() && hasAnyRole(['CENTER_STAFF', 'CENTER_OWNER', 'DRIVER', 'SUPER_ADMIN', 'OPERATIONS_MANAGER']);
      allow create: if isSignedIn() && hasAnyRole(['CENTER_STAFF', 'CENTER_OWNER', 'DRIVER', 'SUPER_ADMIN', 'OPERATIONS_MANAGER']);
    }

    match /scanLogs/{logId} {
      allow read: if isSignedIn() && hasAnyRole(['CENTER_STAFF', 'CENTER_OWNER', 'DRIVER', 'SUPER_ADMIN', 'OPERATIONS_MANAGER']);
      allow create: if isSignedIn() && hasAnyRole(['CENTER_STAFF', 'CENTER_OWNER', 'DRIVER', 'SUPER_ADMIN', 'OPERATIONS_MANAGER']);
    }

    match /bulk_intake_sessions/{sessionId} {
      allow read: if isSignedIn() && (
        hasAnyRole(['CENTER_STAFF', 'CENTER_OWNER', 'SUPER_ADMIN', 'OPERATIONS_MANAGER', 'POINT_ADMIN', 'POINT_STAFF']) ||
        (hasRole('MERCHANT') && resource.data.merchantId == request.auth.uid)
      );
      allow create, update: if isSignedIn() && hasAnyRole(['CENTER_STAFF', 'CENTER_OWNER', 'SUPER_ADMIN', 'OPERATIONS_MANAGER', 'POINT_ADMIN', 'POINT_STAFF']);
    }

    // PHASE A2 (corrected): wesabiHubPoints stores physical Hub/PUDO point location
    // records (name, address, GPS, status, trust score) -- NOT a points/rewards ledger.
    // A hub owner may create and edit their own hub's basic details, but can never:
    //  - set/approve their own hub's status (admin-only, prevents self-approval)
    //  - write their own trustScore/rating/starRating/totalPoints/rankings (system/admin only)
    //  - touch another owner's hub at all
    match /wesabiHubPoints/{hubId} {
      allow read, list: if isSignedIn();

      allow create: if isPlatformAdmin()
                    || (isSignedIn() && hasRole('CENTER_OWNER') &&
                        incoming().ownerId == request.auth.uid &&
                        incoming().status == 'PENDING');

      allow update: if isPlatformAdmin()
                    || (isSignedIn() && existing().ownerId == request.auth.uid &&
                        incoming().ownerId == request.auth.uid &&
                        incoming().diff(existing()).affectedKeys().hasOnly([
                          'name', 'address', 'city', 'state', 'lga', 'country',
                          'type', 'contactPhone', 'operatingHours', 'services',
                          'location', 'gps', 'updatedAt'
                        ]));

      allow delete: if isPlatformAdmin()
                    || (isSignedIn() && existing().ownerId == request.auth.uid);
    }

    match /pointEmployees/{empId} {
      allow read: if isSignedIn();
      allow write: if isAdmin() || hasRole('CENTER_OWNER') || hasRole('OPERATIONS_MANAGER');
    }

    match /merchantBusinesses/{busId} {
      allow read: if true;
      allow create: if isSignedIn() && incoming().merchantId == request.auth.uid;
      allow update: if isSignedIn() && (resource.data.merchantId == request.auth.uid || isAdmin());
    }

    match /logisticsCompanies/{compId} {
      allow read: if true;
      allow create: if isSignedIn() && incoming().ownerId == request.auth.uid;
      allow update: if (isOwner(existing().ownerId) && incoming().diff(existing()).affectedKeys().hasOnly(['name', 'fleetSize', 'serviceAreas']))
                    || isAdmin();
    }

    match /drivers/{driverId} {
      allow read: if isSignedIn();
      allow create: if hasRole('LOGISTICS_OWNER') || isAdmin();
      allow update: if isOwner(driverId) || hasRole('LOGISTICS_OWNER') || isAdmin();
    }

    match /wallets/{userId} {
      allow read: if isOwner(userId) || hasAnyRole(['SUPER_ADMIN', 'FINANCE_OFFICER']);
      allow write: if isAdmin() || hasRole('FINANCE_OFFICER');
    }

    match /transactions/{txId} {
      allow read: if isSignedIn() && (isOwner(existing().walletId) || hasAnyRole(['SUPER_ADMIN', 'FINANCE_OFFICER']));
      allow write: if false; // System / Admin SDK only
    }

    match /verificationRequests/{reqId} {
      allow read: if isOwner(existing().userId) || hasAnyRole(['SUPER_ADMIN', 'VERIFICATION_OFFICER']);
      allow create: if isSignedIn() && incoming().userId == request.auth.uid;
      allow update: if hasAnyRole(['SUPER_ADMIN', 'VERIFICATION_OFFICER']);
    }

    match /idVerifications/{verificationId} {
      allow read: if isSignedIn() && (resource.data.userId == request.auth.uid || isStaff());
      allow create: if isSignedIn() && incoming().userId == request.auth.uid;
      allow update: if isSignedIn() && isStaff();
      allow delete: if isSignedIn() && isStaff();
    }

    match /notifications/{noteId} {
      allow get, update: if isSignedIn() && resource.data.userId == request.auth.uid;
      allow list: if isSignedIn() && (request.auth.uid != null);
      allow create: if isSignedIn();
    }

    // PHASE A2: Audit logs are immutable records created only by server/Admin SDK.
    match /auditLogs/{logId} {
      allow get: if isAdmin() || hasRole('OPERATIONS_MANAGER') || (isSignedIn() && resource.data.userId == request.auth.uid);
      allow list: if isAdmin() || hasRole('OPERATIONS_MANAGER') || (isSignedIn() && resource.data.userId == request.auth.uid);
      // CRITICAL: No client-side creation. Audit logs are server/Admin SDK only.
      allow create: if false;
      allow update, delete: if false; // Read-only & Tamper-resistant!
    }

    // PHASE A2: System settings are admin-controlled. Only SUPER_ADMIN/OPERATIONS_ADMIN can modify.
    match /systemSettings/{document=**} {
      allow read: if true; // Public read for non-sensitive config
      allow write: if isAdmin();
    }

    match /adminRoles/{roleId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }

    match /brandAssets/{brandId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /conversations/{convId} {
      allow read: if isSignedIn() && (
        resource.data.buyerId == request.auth.uid ||
        resource.data.sellerId == request.auth.uid ||
        (resource.data.participants != null && request.auth.uid in resource.data.participants) ||
        isStaff()
      );
      allow create: if isSignedIn() && (incoming().buyerId == request.auth.uid || incoming().sellerId == request.auth.uid || (incoming().participants != null && request.auth.uid in incoming().participants));
      allow update: if isSignedIn() && (
        ((resource.data.buyerId == request.auth.uid || resource.data.sellerId == request.auth.uid || (resource.data.participants != null && request.auth.uid in resource.data.participants))
         && (existing().isDisputed != isOwner(userId) || incoming().isDisputed == true))
        || isStaff()
      );
    }

    match /conversations/{convId}/messages/{msgId} {
      allow read: if isSignedIn() && exists(/databases/$(database)/documents/conversations/$(convId)) && (
        get(/databases/$(database)/documents/conversations/$(convId)).data.buyerId == request.auth.uid ||
        get(/databases/$(database)/documents/conversations/$(convId)).data.sellerId == request.auth.uid ||
        (get(/databases/$(database)/documents/conversations/$(convId)).data.participants != null && request.auth.uid in get(/databases/$(database)/documents/conversations/$(convId)).data.participants) ||
        isStaff()
      );
      allow create: if isSignedIn() && exists(/databases/$(database)/documents/conversations/$(convId)) && (
        (get(/databases/$(database)/documents/conversations/$(convId)).data.buyerId == request.auth.uid ||
         get(/databases/$(database)/documents/conversations/$(convId)).data.sellerId == request.auth.uid ||
         (get(/databases/$(database)/documents/conversations/$(convId)).data.participants != null && request.auth.uid in get(/databases/$(database)/documents/conversations/$(convId)).data.participants))
        && get(/databases/$(database)/documents/conversations/$(convId)).data.isDisputed != true
      );
      allow update, delete: if false; // Messages are strictly immutable
    }

    match /disputes/{disputeId} {
      allow list: if isSignedIn() && (
        isStaff() ||
        resource.data.initiatorId == request.auth.uid
      );
      allow get: if isSignedIn() && (
        resource.data.initiatorId == request.auth.uid ||
        (exists(/databases/$(database)/documents/conversations/$(resource.data.conversationId)) && (
          get(/databases/$(database)/documents/conversations/$(resource.data.conversationId)).data.buyerId == request.auth.uid ||
          get(/databases/$(database)/documents/conversations/$(resource.data.conversationId)).data.sellerId == request.auth.uid
        )) ||
        isStaff()
      );
      allow create: if isSignedIn();
      allow update: if isSignedIn() && isStaff();
    }

    match /support_tickets/{ticketId} {
      allow list: if isSignedIn() && (isStaff() || resource.data.userId == request.auth.uid);
      allow get: if isSignedIn() && (resource.data.userId == request.auth.uid || isStaff());
      allow create: if isSignedIn() && incoming().userId == request.auth.uid;
      allow update: if isSignedIn() && (existing().userId == request.auth.uid || isStaff());
    }

    match /knowledgeArticles/{articleId} {
      allow read: if true;
      allow write: if isPlatformAdmin();
    }

    match /faqCategories/{catId} {
      allow read: if true;
      allow write: if isPlatformAdmin();
    }

    match /countries/{countryId} {
      allow read: if true;
      allow write: if isAdmin() || hasRole('OPERATIONS_MANAGER');
    }

    match /pricingRules/{ruleId} {
      allow read: if true;
      allow write: if isAdmin() || hasRole('FINANCE_OFFICER');
    }

    match /commissionRules/{ruleId} {
      allow read: if true;
      allow write: if isAdmin() || hasRole('FINANCE_OFFICER');
    }

    match /commissionRecords/{recordId} {
      allow read: if isSignedIn() && (
        isAdmin() ||
        hasRole('FINANCE_OFFICER') ||
        resource.data.centreId == request.auth.uid ||
        resource.data.merchantId == request.auth.uid ||
        (exists(/databases/$(database)/documents/wesabiHubPoints/$(resource.data.centreId)) &&
         get(/databases/$(database)/documents/wesabiHubPoints/$(resource.data.centreId)).data.ownerId == request.auth.uid)
      );
      allow write: if false; // System only (Backend API via admin SDK)
    }

    match /commissionAuditLogs/{logId} {
      allow read: if isAdmin() || hasRole('FINANCE_OFFICER');
      allow create: if isAdmin() || hasRole('FINANCE_OFFICER');
    }

    match /serviceZones/{zoneId} {
      allow read: if true;
      allow write: if isAdmin() || hasRole('OPERATIONS_MANAGER');
    }

    match /promotionRules/{promoId} {
      allow read: if true;
      allow write: if isAdmin() || hasRole('OPERATIONS_MANAGER');
    }

    match /paymentProtections/{protectionId} {
      allow read: if isSignedIn() && (
        resource.data.customerId == request.auth.uid ||
        resource.data.merchantId == request.auth.uid ||
        isStaff()
      );
      allow create: if isSignedIn() && (
        incoming().customerId == request.auth.uid ||
        isStaff()
      );
      allow update: if isSignedIn() && (
        // Terminal state lock: Cannot edit finalized payments
        !(existing().status in ['PAYMENT_RELEASED', 'REFUND_APPROVED', 'PARTIAL_REFUND_APPROVED', 'TRANSACTION_CLOSED']) && (
          // Allow customer or merchant to raise a dispute or adjust metadata
          ((existing().customerId == request.auth.uid || existing().merchantId == request.auth.uid) &&
           incoming().diff(existing()).affectedKeys().hasOnly(['status', 'disputeId', 'metadata'])) ||
          // Allow admins to resolve or manage status
          isStaff()
        )
      );
      allow delete: if false; // Secure payments are permanent records
    }

    match /paymentProtectionSettings/{settingsId} {
      allow read: if true;
      allow write: if isSignedIn() && hasAnyRole(['SUPER_ADMIN', 'FINANCE_OFFICER']);
    }

    match /returnRequests/{requestId} {
      allow read: if isSignedIn() && (
        resource.data.customerUserId == request.auth.uid ||
        resource.data.merchantId == request.auth.uid ||
        hasAnyRole(['SUPER_ADMIN', 'SUPPORT_OFFICER', 'OPERATIONS_MANAGER'])
      );
      allow create: if isSignedIn() && incoming().customerUserId == request.auth.uid;
      allow update: if isSignedIn() && (
        (existing().customerUserId == request.auth.uid && incoming().diff(existing()).affectedKeys().hasOnly(['description', 'photos', 'status'])) ||
        (existing().merchantId == request.auth.uid && incoming().diff(existing()).affectedKeys().hasOnly(['status'])) ||
        hasAnyRole(['SUPER_ADMIN', 'SUPPORT_OFFICER', 'OPERATIONS_MANAGER'])
      );
      allow delete: if false;
    }

    // New rules added to prevent "Missing or insufficient permissions"
    // PHASE A2: Dispute categories are system configuration. Admin-only writes.
    match /disputeCategories/{catId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /webhookLogs/{logId} {
      allow read: if isSignedIn() && (resource.data.userId == request.auth.uid || isPlatformAdmin());
      allow write: if isSignedIn() && (incoming().userId == request.auth.uid || isPlatformAdmin());
    }

    match /webhook_logs/{logId} {
      allow read: if isSignedIn() && (resource.data.userId == request.auth.uid || isPlatformAdmin());
      allow write: if isSignedIn() && (incoming().userId == request.auth.uid || isPlatformAdmin());
    }

    match /apiLogs/{logId} {
      allow read: if isSignedIn() && (resource.data.userId == request.auth.uid || isPlatformAdmin());
      allow write: if isSignedIn() && (incoming().userId == request.auth.uid || isPlatformAdmin());
    }

    match /api_logs/{logId} {
      allow read: if isSignedIn() && (resource.data.userId == request.auth.uid || isPlatformAdmin());
      allow write: if isSignedIn() && (incoming().userId == request.auth.uid || isPlatformAdmin());
    }

    match /faqs/{faqId} {
      allow read: if true;
      allow write: if isSignedIn() && isPlatformAdmin();
    }

    match /page_content/{id} {
      allow read: if true;
      allow write: if isSignedIn() && isPlatformAdmin();
    }

    match /announcements/{id} {
      allow read: if true;
      allow write: if isSignedIn() && isPlatformAdmin();
    }

    // PHASE A2: Advertisements are managed by admins. Prevent unauthorized client writes.
    match /advertisements/{allPaths=**} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /complaints/{id} {
      allow read: if isSignedIn() && (isOwner(resource.data.userId) || isStaff());
      allow create: if isSignedIn();
      allow update: if isSignedIn() && (isOwner(existing().userId) || isStaff());
    }

    match /courseCategories/{id} {
      allow read: if true;
      allow write: if isSignedIn() && isPlatformAdmin();
    }

    match /courses/{id} {
      allow read: if true;
      allow write: if isSignedIn() && isPlatformAdmin();
    }

    match /invitations/{id} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update: if isSignedIn();
    }

    match /staffInvitations/{id} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update: if isSignedIn();
    }

    // PHASE A2: Policy versions are system configuration. Admin-only writes.
